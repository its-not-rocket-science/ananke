Quick start

import { createWorld, stepWorld, q, SCALE } from "@its-not-rocket-science/ananke";

const world = createWorld({ worldSeed: 42, entities: [] });
const next  = stepWorld(world, []);
console.log("tick:", next.tick);
// Tier 2 combat subpath
import { resolveHit } from "@its-not-rocket-science/ananke/sim/combat";
import { exposeToDisease } from "@its-not-rocket-science/ananke/sim/disease";
import { computeHazardExposure, CAMPFIRE } from "@its-not-rocket-science/ananke/sim/hazard";
